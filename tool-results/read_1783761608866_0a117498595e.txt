    25→function json(body: unknown, status = 200): Response {
    26→  return new Response(JSON.stringify(body), {
    27→    status,
    28→    headers: { "Content-Type": "application/json" },
    29→  });
    30→}
    31→
    32→function ok(data: unknown, extra?: Record<string, unknown>, status = 200): Response {
    33→  return json({ success: true, data, ...(extra || {}) }, status);
    34→}
    35→
    36→function fail(error: string, status = 500): Response {
    37→  return json({ success: false, error }, status);
    38→}

    40→// ----------------------------------------------------------------------------
    41→// URL / body parsing helpers
    42→// ----------------------------------------------------------------------------
    43→
    44→interface ParsedRequest {
    45→  pathname: string;
    46→  search: URLSearchParams;
    47→  method: string;
    48→  body: any;
    49→}
    50→
    51→async function parseRequest(
    52→  input: RequestInfo | URL,
    53→  init?: RequestInit
    54→): Promise<ParsedRequest> {
    55→  let urlStr: string;
    56→  let method = (init?.method || "GET").toUpperCase();
    57→  let body: any = undefined;
    58→
    59→  if (input instanceof Request) {
    60→    urlStr = input.url;
    61→    if (!init) {
    62→      method = (input.method || "GET").toUpperCase();
    63→      try {
    64→        body = await input.json();
    65→      } catch {
    66→        try {
    67→          const text = await input.text();
    68→          body = text ? safeJson(text) : undefined;
    69→        } catch {
    70→          /* ignore */
    71→        }
    72→      }
    73→    }
    74→  } else if (input instanceof URL) {
    75→    urlStr = input.toString();
    76→  } else {
    77→    urlStr = input as string;
    78→  }
    79→
    80→  if (init?.body && body === undefined) {
    81→    if (typeof init.body === "string") {
    82→      body = safeJson(init.body);
    83→    } else if (init.body instanceof FormData) {
    84→      const obj: Record<string, unknown> = {};
    85→      init.body.forEach((value, key) => {
    86→        obj[key] = value;
    87→      });
    88→      body = obj;
    89→    } else if (init.body instanceof Blob) {
    90→      try {
    91→        body = safeJson(await init.body.text());
    92→      } catch {
    93→        body = undefined;
    94→      }
    95→    }
    96→  }
    97→
    98→  // Strip the protocol + host if present so we only route on the path.
    99→  let url: URL;
   100→  try {
   101→    url = new URL(urlStr, typeof window !== "undefined" ? window.location.origin : "http://localhost");
   102→  } catch {
   103→    url = new URL("http://localhost" + urlStr);
   104→  }
   105→
   106→  return {
   107→    pathname: url.pathname,
   108→    search: url.searchParams,
   109→    method,
   110→    body,
   111→  };
   112→}
   113→
   114→function safeJson(text: string): any {
   115→  try {
   116→    return JSON.parse(text);
   117→  } catch {
   118→    return text;
   119→  }
   120→}

   141→  return (
   142→    a.getFullYear() === b.getFullYear() &&
   143→    a.getMonth() === b.getMonth() &&
   144→    a.getDate() === b.getDate()
   145→  );

   156→interface CrudConfig {
   157→  collection: string;
   158→  softDelete?: boolean;
   159→  /** Default orderBy applied to GET list. */
   160→  orderBy?: (items: any[]) => any[];
   161→  /** Extra filters based on query params. */
   162→  filter?: (items: any[], search: URLSearchParams) => any[];
   163→  /** Transform the GET payload (e.g. add stats). */
   164→  transformGet?: (items: any[], search: URLSearchParams) => Record<string, unknown>;
   165→  /** Custom create payload (defaults to the body). */
   166→  createPayload?: (body: any) => any;
   167→  /** Activity log message on create. */
   168→  createLog?: (item: any) => string;
   169→  /** Validation: returns an error string or null. */
   170→  validateCreate?: (body: any) => string | null;
   171→  /** Validation for updates. */
   172→  validateUpdate?: (body: any) => string | null;
   173→  /** Transform PUT body before applying. */
   174→  transformUpdate?: (body: any) => any;
   175→  /** Status code for POST. */
   176→  createStatus?: number;
   177→}
   178→
   179→function crudHandler(cfg: CrudConfig) {
   180→  return async (req: ParsedRequest): Promise<Response> => {
   181→    const items = db.getCollection(cfg.collection);
   182→
   183→    if (req.method === "GET") {
   184→      let visible = items.filter((i) => i.deletedAt == null);
   185→      if (cfg.filter) visible = cfg.filter(visible, req.search);
   186→      if (cfg.orderBy) visible = cfg.orderBy(visible);
   187→      const extra = cfg.transformGet ? cfg.transformGet(visible, req.search) : {};
   188→      return ok(visible, extra);
   189→    }
   190→
   191→    if (req.method === "POST") {
   192→      if (cfg.validateCreate) {
   193→        const err = cfg.validateCreate(req.body);
   194→        if (err) return fail(err, 400);
   195→      }
   196→      const payload = cfg.createPayload ? cfg.createPayload(req.body) : req.body;
   197→      const created = db.insert(cfg.collection, payload);
   198→      if (cfg.createLog) db.logActivity("create", cfg.collection, cfg.createLog(created));
   199→      return ok(created, undefined, cfg.createStatus ?? 201);
   200→    }
   201→
   202→    if (req.method === "PUT") {
   203→      if (cfg.validateUpdate) {
   204→        const err = cfg.validateUpdate(req.body);
   205→        if (err) return fail(err, 400);
   206→      }
   207→      const { id, ...rest } = req.body || {};
   208→      if (!id) return fail("المعرف مطلوب", 400);
   209→      const patch = cfg.transformUpdate ? cfg.transformUpdate(rest) : rest;
   210→      const updated = db.update(cfg.collection, id, patch);
   211→      if (!updated) return fail("العنصر غير موجود", 404);
   212→      return ok(updated);
   213→    }
   214→
   215→    if (req.method === "DELETE") {
   216→      const id = req.search.get("id");
   217→      const force = req.search.get("force") === "true";
   218→      if (!id) return fail("المعرف مطلوب", 400);
   219→      if (cfg.softDelete && !force) {
   220→        const okFlag = db.softDelete(cfg.collection, id);
   221→        if (!okFlag) return fail("العنصر غير موجود", 404);
   222→      } else {
   223→        const okFlag = db.remove(cfg.collection, id);
   224→        if (!okFlag) return fail("العنصر غير موجود", 404);
   225→      }
   226→      return json({ success: true });
   227→    }
   228→
   229→    return fail("الطريقة غير مدعومة", 405);
   230→  };
   231→}

   237→// ---- /api/calllogs ---------------------------------------------------------
   238→async function calllogsRoute(req: ParsedRequest): Promise<Response> {
   239→  if (req.method === "GET") {
   240→    const logs = db
   241→      .getCollection("callLogs")
   242→      .sort((a, b) => new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime())
   243→      .slice(0, 50);
   244→    // Enrich with contact info (like Prisma's `include: { contact: true }`)
   245→    const contacts = db.getCollection("contacts");
   246→    const data = logs.map((l) => ({
   247→      ...l,
   248→      contact: l.contactId ? contacts.find((c) => c.id === l.contactId) || null : null,
   249→    }));
   250→    return ok(data);
   251→  }
   252→  if (req.method === "POST") {
   253→    const { contactId, name, phone, type, direction, note } = req.body || {};
   254→    if (!name || !phone) return fail("الاسم والهاتف مطلوبان", 400);
   255→    const log = db.insert("callLogs", {
   256→      contactId: contactId || null,
   257→      name,
   258→      phone,
   259→      type: type || "call",
   260→      direction: direction || "outgoing",
   261→      note: note || null,
   262→    });
   263→    db.logActivity("create", "calllog", `تم تسجيل مكالمة: ${name}`);
   264→    return ok(log, 201);
   265→  }
   266→  if (req.method === "DELETE") {
   267→    const id = req.search.get("id");
   268→    if (!id) {
   269→      db.setCollection("callLogs", []);
   270→      return ok(null);
   271→    }
   272→    db.remove("callLogs", id);
   273→    return ok(null);
   274→  }
   275→  return fail("Method not allowed", 405);
   276→}

   362→// ---- /api/tasks -------------------------------------------------------------
   363→async function tasksRoute(req: ParsedRequest): Promise<Response> {
   364→  const items = db.getCollection("tasks");
   365→
   366→  if (req.method === "GET") {
   367→    const status = req.search.get("status");
   368→    const category = req.search.get("category");
   369→    const priority = req.search.get("priority");
   370→    let visible = items.filter((i) => i.deletedAt == null);
   371→    if (status) visible = visible.filter((i) => i.status === status);
   372→    if (category) visible = visible.filter((i) => i.category === category);
   373→    if (priority) visible = visible.filter((i) => i.priority === priority);
   374→
   375→    const projects = db.getCollection("projects");
   376→    const priorityOrder: Record<string, number> = { high: 0, medium: 1, low: 2 };
   377→    visible.sort((a, b) => {
   378→      const pa = priorityOrder[a.priority] ?? 99;
   379→      const pb = priorityOrder[b.priority] ?? 99;
   380→      if (pa !== pb) return pa - pb;
   381→      return new Date(b.createdAt).getTime() - new Date(a.createdAt).getTime();
   382→    });
   383→    const data = visible.map((t) => ({
   384→      ...t,
   385→      project: projects.find((p) => p.id === t.projectId) || null,
   386→    }));
   387→
   388→    const stats = {
   389→      total: visible.length,
   390→      todo: visible.filter((t) => t.status === "todo").length,
   391→      doing: visible.filter((t) => t.status === "doing").length,
   392→      done: visible.filter((t) => t.status === "done").length,
   393→      high: visible.filter((t) => t.priority === "high").length,
   394→      overdue: visible.filter(
   395→        (t) => t.status !== "done" && t.dueDate && new Date(t.dueDate) < new Date()
   396→      ).length,
   397→    };
   398→    return ok(data, { stats });
   399→  }
   400→
   401→  if (req.method === "POST") {
   402→    if (!req.body?.title) return fail("العنوان مطلوب", 400);
   403→    const payload = {
   404→      title: req.body.title,
   405→      description: req.body.description || null,
   406→      status: req.body.status || "todo",
   407→      priority: req.body.priority || "medium",
   408→      category: req.body.category || "general",
   409→      dueDate: req.body.dueDate ? new Date(req.body.dueDate).toISOString() : null,
   410→      projectId: req.body.projectId || null,
   411→      deletedAt: null,
   412→    };
   413→    const created = db.insert("tasks", payload);
   414→    db.logActivity("create", "task", `أضيف مهمة: ${created.title}`);
   415→    return ok(created, undefined, 201);
   416→  }
   417→
   418→  if (req.method === "PUT") {
   419→    const { id, ...rest } = req.body || {};
   420→    if (!id) return fail("المعرف مطلوب", 400);
   421→    const patch = { ...rest };
   422→    if (patch.dueDate) patch.dueDate = new Date(patch.dueDate).toISOString();
   423→    const updated = db.update("tasks", id, patch);
   424→    if (!updated) return fail("العنصر غير موجود", 404);
   425→    return ok(updated);
   426→  }
   427→
   428→  if (req.method === "DELETE") {
   429→    const id = req.search.get("id");
   430→    const force = req.search.get("force") === "true";
   431→    if (!id) return fail("المعرف مطلوب", 400);
   432→    if (force) db.remove("tasks", id);
   433→    else db.softDelete("tasks", id);
   434→    return json({ success: true });
   435→  }
   436→
   437→  return fail("الطريقة غير مدعومة", 405);
   438→}

   501→// ---- /api/expenses ----------------------------------------------------------
   502→async function expensesRoute(req: ParsedRequest): Promise<Response> {
   503→  const items = db.getCollection("expenses");
   504→
   505→  if (req.method === "GET") {
   506→    const category = req.search.get("category");
   507→    const from = req.search.get("from");
   508→    const to = req.search.get("to");
   509→    let visible = items.filter((i) => i.deletedAt == null);
   510→    if (category) visible = visible.filter((i) => i.category === category);
   511→    if (from) {
   512→      const f = new Date(from).getTime();
   513→      visible = visible.filter((i) => new Date(i.date).getTime() >= f);
   514→    }
   515→    if (to) {
   516→      const t = new Date(to).getTime();
   517→      visible = visible.filter((i) => new Date(i.date).getTime() <= t);
   518→    }
   519→    visible.sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime());
   520→
   521→    const totalSYP = visible.filter((e) => e.currency === "syp").reduce((s, e) => s + e.amount, 0);
   522→    const totalUSD = visible.filter((e) => e.currency === "usd").reduce((s, e) => s + e.amount, 0);
   523→
   524→    const byCategory: Record<string, { syp: number; usd: number; count: number }> = {};
   525→    for (const e of visible) {
   526→      if (!byCategory[e.category]) byCategory[e.category] = { syp: 0, usd: 0, count: 0 };
   527→      if (e.currency === "syp") byCategory[e.category].syp += e.amount;
   528→      else byCategory[e.category].usd += e.amount;
   529→      byCategory[e.category].count++;
   530→    }
   531→
   532→    return ok(visible, {
   533→      stats: { totalSYP, totalUSD, count: visible.length, byCategory },
   534→    });
   535→  }
   536→
   537→  if (req.method === "POST") {
   538→    if (!req.body?.amount) return fail("المبلغ مطلوب", 400);
   539→    const payload = {
   540→      amount: Number(req.body.amount),
   541→      currency: req.body.currency || "syp",
   542→      category: req.body.category || "general",
   543→      description: req.body.description || null,
   544→      date: req.body.date ? new Date(req.body.date).toISOString() : nowISO(),
   545→      deletedAt: null,
   546→    };

  1418→function computeStreak(logs: Array<{ date: string }>): number {
  1419→  if (!logs.length) return 0;
  1420→  const sorted = logs
  1421→    .map((l) => {
  1422→      const d = new Date(l.date);
  1423→      d.setHours(0, 0, 0, 0);
  1424→      return d.getTime();
  1425→    })
  1426→    .sort((a, b) => b - a);
  1427→  const today = startOfToday();
  1428→  let streak = 0;
  1429→  let cursor = today.getTime();
  1430→  if (sorted[0] !== cursor) {
  1431→    const yesterday = cursor - 86400000;
  1432→    if (sorted[0] !== yesterday) return 0;
  1433→    cursor = yesterday;
  1434→  }
  1435→  for (const ts of sorted) {
  1436→    if (ts === cursor) {
  1437→      streak++;
  1438→      cursor -= 86400000;
  1439→    } else if (ts < cursor) {
  1440→      break;
  1441→    }
  1442→  }
  1443→  return streak;
  1444→}
